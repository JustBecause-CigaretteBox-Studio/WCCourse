第 1 小节

需求分析

我们要做一个最小的「学生管理系统」。在动手写代码之前，先想清楚两个问题：

  • 学生有什么信息？ 比如姓名、年龄、若干科目的成绩。
  • 系统需要支持什么操作？ 比如添加学生、录入成绩、查询某个学生、列出所有人。

把这两个问题回答清楚，程序的结构就自然浮现出来了：每个学生可以封装成一个 Student 类，整个系统再用一个 StudentSystem 类来管理所有学生。

第 2 小节

设计 Student 类

单个学生只需要关心自己的数据和行为。我们把成绩存成列表vector，并提供添加成绩和计算平均分的方法。

student.py
class Student:
    def __init__(self, name, age):
        self.name = name
        self.age = age
        self.scores = []

    def add_score(self, subject, score):
        self.scores.append((subject, score))

    def average(self):
        if not self.scores:
            return 0
        total = sum(s for _, s in self.scores)
        return total / len(self.scores)

    def show(self):
        print(self.name, self.age, "岁，平均分：", self.average())

xiaoming.add_score("数学", 90)
xiaoming.add_score("语文", 85)
xiaoming.show()
student.cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Student {
public:
    string name;
    int age;
    vector<pair<string, int>> scores;

    Student(string n, int a) {
        name = n;
        age = a;
    }

    void addScore(string subject, int score) {
        scores.push_back({subject, score});
    }

    double average() {
        if (scores.empty()) return 0;
        int total = 0;
        for (auto &p : scores) total += p.second;
        return (double)total / scores.size();
    }

    void show() {
        cout << name << " " << age << " 岁，平均分：" << average() << endl;
    }
};

int main() {
    Student xiaoming("小明", 12);
    xiaoming.addScore("数学", 90);
    xiaoming.addScore("语文", 85);
    xiaoming.show();
    return 0;
}

这里 scores 是一个列表，列表里每个元素是 (科目, 分数) 的元组vector<pair<string, int>>，每个元素是一对 {科目, 分数}。这样设计的好处是：科目和分数始终成对出现，不会弄混。

第 3 小节

设计 StudentSystem 类

系统类的职责是管理一群学生。它不需要知道成绩的计算细节，而是调用每个 Student 对象自己的方法。这种「各司其职」的设计正是面向对象的优势。

student_system.py
class StudentSystem:
    def __init__(self):
        self.students = []

    def add_student(self, student):
        self.students.append(student)

    def find(self, name):
        for s in self.students:
            if s.name == name:
                return s
        return None

    def list_all(self):
        for s in self.students:
            s.show()

sys = StudentSystem()
sys.add_student(Student("小明", 12))
sys.add_student(Student("小红", 11))

s = sys.find("小明")
if s:
    s.add_score("数学", 95)

sys.list_all()
student_system.cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

class Student {
public:
    string name;
    int age;
    vector<pair<string, int>> scores;

    Student(string n, int a) : name(n), age(a) {}

    void addScore(string subject, int score) {
        scores.push_back({subject, score});
    }

    double average() {
        if (scores.empty()) return 0;
        int total = 0;
        for (auto &p : scores) total += p.second;
        return (double)total / scores.size();
    }

    void show() {
        cout << name << " " << age << " 岁，平均分：" << average() << endl;
    }
};

class StudentSystem {
public:
    vector<Student> students;

    void addStudent(Student student) {
        students.push_back(student);
    }

    Student* find(string name) {
        for (auto &s : students) {
            if (s.name == name) return &s;
        }
        return nullptr;
    }

    void listAll() {
        for (auto &s : students) s.show();
    }
};

int main() {
    StudentSystem sys;
    sys.addStudent(Student("小明", 12));
    sys.addStudent(Student("小红", 11));

    Student* s = sys.find("小明");
    if (s) s->addScore("数学", 95);

    sys.listAll();
    return 0;
}

注意 find 方法返回的是 Student 对象本身。找到之后，我们可以直接在这个对象上调用 add_scoreaddScore，而不需要系统类去处理成绩细节。

#include <iostream>
#include <iomanip>
#include <string>
#include <vector>
using namespace std;

class Student {
public:
    string name;
    int age;
    vector<pair<string, int>> scores;

    Student(string n, int a) : name(n), age(a) {}

    void addScore(string subject, int score) {
        scores.push_back({subject, score});
    }

    double average() {
        if (scores.empty()) return 0;
        int total = 0;
        for (auto &p : scores) total += p.second;
        return (double)total / scores.size();
    }

    void show() {
        cout << name << "（" << age << " 岁）平均分："
             << fixed << setprecision(1) << average() << endl;
    }
};

class StudentSystem {
public:
    vector<Student> students;

    void addStudent(string name, int age) {
        students.push_back(Student(name, age));
    }

    Student* find(string name) {
        for (auto &s : students) {
            if (s.name == name) return &s;
        }
        return nullptr;
    }

    void listAll() {
        for (auto &s : students) s.show();
    }
};

int main() {
    StudentSystem sys;
    sys.addStudent("小明", 12);
    sys.addStudent("小红", 11);

    sys.find("小明")->addScore("数学", 92);
    sys.find("小明")->addScore("语文", 88);
    sys.find("小红")->addScore("数学", 97);

    sys.listAll();
    return 0;
}
