第 1 小节

加载与显示图片

纯色方块虽然够用，但真正的游戏需要图片。Pygame 用 pygame.image.load() 加载图片文件，再用 blit() 贴到屏幕上。

注意图片要放在和 Python 文件同一个文件夹里，或者提供完整路径。支持的格式包括 PNG、JPG、GIF 等。

image_load.py
import pygame

pygame.init()
screen = pygame.display.set_mode((800, 600))
clock = pygame.time.Clock()

# 假设同目录下有一张 player.png
player_img = pygame.image.load("player.png").convert_alpha()
player_rect = player_img.get_rect()
player_rect.center = (400, 300)

running = True
while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

    screen.fill((40, 40, 50))
    screen.blit(player_img, player_rect)
    pygame.display.flip()
    clock.tick(60)

pygame.quit()

convert_alpha() 会保留图片的透明背景，适合角色、道具等需要透明通道的素材。如果没有透明需求，可以用 convert()，运行更快。

第 2 小节

播放音效

音效能让游戏瞬间变得生动。Pygame 的 pygame.mixer 模块负责播放声音。需要先初始化，然后加载音频文件再播放。

sound.py
import pygame

pygame.init()
pygame.mixer.init()
screen = pygame.display.set_mode((800, 600))
clock = pygame.time.Clock()

jump_sound = pygame.mixer.Sound("jump.wav")

running = True
while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_SPACE:
                jump_sound.play()

    screen.fill((30, 30, 30))
    pygame.display.flip()
    clock.tick(60)

pygame.quit()

把 jump.wav 换成你自己的音频文件即可。WAV 和 OGG 格式支持得最好。背景音乐通常用 pygame.mixer.music.load() 和 .play() 播放。

第 3 小节

文字与分数显示

游戏需要告诉玩家分数、生命值、提示信息。Pygame 用 pygame.font 渲染文字，把文字转成 Surface 后 blit() 到屏幕上。

text.py
import pygame

pygame.init()
screen = pygame.display.set_mode((800, 600))
clock = pygame.time.Clock()
font = pygame.font.SysFont("simhei", 36)

score = 0
running = True
while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        if event.type == pygame.MOUSEBUTTONDOWN:
            score += 10

    text_surface = font.render("分数: " + str(score), True, (255, 255, 255))

    screen.fill((25, 25, 25))
    screen.blit(text_surface, (20, 20))
    pygame.display.flip()
    clock.tick(60)

pygame.quit()

font.render(文字, 是否抗锯齿, 颜色) 返回一个 Surface。你可以把它当成普通图片来贴。中文字体推荐用系统自带的「simhei」或「microsoftyahei」。

第 4 小节

综合实战：给游戏加皮肤和音效

假设我们有一个简单的躲避游戏：玩家控制飞机躲陨石。现在给它加上图片素材、碰撞音效和分数显示。

polished.py
import pygame
import random

pygame.init()
pygame.mixer.init()
screen = pygame.display.set_mode((800, 600))
clock = pygame.time.Clock()
font = pygame.font.SysFont("simhei", 32)

# 请把 player.png 和 hit.wav 放到同目录
player = pygame.image.load("player.png").convert_alpha()
player_rect = player.get_rect()
player_rect.center = (400, 500)

hit_sound = pygame.mixer.Sound("hit.wav")
meteors = [pygame.Rect(random.randint(0, 760), -50, 40, 40) for _ in range(5)]
score = 0

running = True
while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

    keys = pygame.key.get_pressed()
    if keys[pygame.K_LEFT]:  player_rect.x -= 6
    if keys[pygame.K_RIGHT]: player_rect.x += 6

    for m in meteors:
        m.y += 4
        if m.top > 600:
            m.y = -50
            m.x = random.randint(0, 760)
            score += 1

    for m in meteors:
        if player_rect.colliderect(m):
            hit_sound.play()
            m.y = -50
            score -= 5

    screen.fill((15, 15, 25))
    screen.blit(player, player_rect)
    for m in meteors:
        pygame.draw.rect(screen, (180, 100, 80), m)

    text = font.render("分数: " + str(score), True, (255, 255, 255))
    screen.blit(text, (20, 20))
    pygame.display.flip()
    clock.tick(60)

pygame.quit()

练习：把陨石也换成图片；给躲过陨石加音效；添加生命值显示。
