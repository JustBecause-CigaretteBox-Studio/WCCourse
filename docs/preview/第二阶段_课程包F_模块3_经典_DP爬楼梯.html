第 1 小节

爬楼梯问题

爬楼梯是动态规划最经典的入门题。问题很简单：一共有 n 级台阶，你每次可以走 1 级或 2 级，问有多少种不同的方法走到第 n 级？

如果你学得很认真，应该会有一种熟悉感。没错，这和之前那个问题是等价的。先来复习一下思路，然后我们会进行一些变化

先从小例子找规律。n=1 时只有 1 种走法；n=2 时可以 1+1 或直接 2，共 2 种。n=3 时可以是 1+1+1、1+2、2+1，共 3 种。n=4 时共有 5 种。看出来了吗？这些数字正好就是斐波那契数列：1, 2, 3, 5, 8, 13...

为什么？因为要到第 n 级，最后一步必然是从第 n-1 级跨 1 步，或者从第 n-2 级跨 2 步。所以 "到 n 级的方法数" 等于 "到 n-1 级的方法数" 加上 "到 n-2 级的方法数"。

climb_stairs.py
# 爬楼梯：递推版
def climb_stairs(n):
    if n <= 2:
        return n
    dp = [0] * (n + 1)
    dp[1] = 1
    dp[2] = 2
    for i in range(3, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]
    return dp[n]

print(climb_stairs(5))   # 8
print(climb_stairs(10))  # 89
climb_stairs.cpp
// 爬楼梯：递推版
#include <iostream>
#include <vector>
using namespace std;

int climb_stairs(int n) {
    if (n <= 2) return n;
    vector<int> dp(n + 1, 0);
    dp[1] = 1;
    dp[2] = 2;
    for (int i = 3; i <= n; ++i) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }
    return dp[n];
}

int main() {
    cout << climb_stairs(5) << endl;   // 8
    cout << climb_stairs(10) << endl;  // 89
    return 0;
}
第 2 小节

状态转移方程

动态规划做题时，最重要的是写出状态转移方程。它描述了"当前状态"如何从"之前的状态"推导出来。

对于爬楼梯，我们定义 dp[i] 表示到达第 i 级台阶的方法数。状态转移方程就是：dp[i] = dp[i-1] + dp[i-2]。边界条件是 dp[1]=1，dp[2]=2。

写状态转移方程的通用套路是：先问自己"最后一步发生了什么"，然后把所有可能的最后一步列出来，分别对应哪些更小的状态。这个方法在背包、线性 DP 等更复杂的问题里同样适用。

state_transition.py
# 爬楼梯：显式写出状态转移
n = 6
dp = [0] * (n + 1)
dp[1] = 1          # 边界
dp[2] = 2          # 边界

# 状态转移：dp[i] = dp[i-1] + dp[i-2]
for i in range(3, n + 1):
    dp[i] = dp[i - 1] + dp[i - 2]

print(dp)          # [0, 1, 2, 3, 5, 8, 13]
print(dp[n])       # 13
state_transition.cpp
// 爬楼梯：显式写出状态转移
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n = 6;
    vector<int> dp(n + 1, 0);
    dp[1] = 1;          // 边界
    dp[2] = 2;          // 边界

    // 状态转移：dp[i] = dp[i-1] + dp[i-2]
    for (int i = 3; i <= n; ++i) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }

    for (int x : dp) cout << x << " ";  // 0 1 2 3 5 8 13
    cout << endl;
    cout << dp[n] << endl;              // 13
    return 0;
}
第 3 小节

空间优化与变形

观察状态转移方程 dp[i] = dp[i-1] + dp[i-2]，你会发现计算 dp[i] 时只需要前两个值。因此我们不需要保存整个数组，只要两个变量滚动更新即可，把空间复杂度从 O(n) 降到 O(1)。

爬楼梯还有很多变形：如果一次可以走 1、2、3 级，方程就变成 dp[i] = dp[i-1] + dp[i-2] + dp[i-3]；如果某些台阶是坏的不能踩，只需要在转移时跳过坏台阶即可。掌握了状态转移，变形就不难。

climb_opt.py
# 空间优化版：只需要两个变量
def climb_stairs_o1(n):
    if n <= 2:
        return n
    prev2, prev1 = 1, 2   # 分别代表 dp[i-2] 和 dp[i-1]
    for _ in range(3, n + 1):
        curr = prev1 + prev2
        prev2, prev1 = prev1, curr
    return prev1

print(climb_stairs_o1(10))  # 89

# 变形：每次可走 1、2、3 级
def climb_three(n):
    if n == 1: return 1
    if n == 2: return 2
    if n == 3: return 4   # 1+1+1, 1+2, 2+1, 3
    a, b, c = 1, 2, 4
    for _ in range(4, n + 1):
        a, b, c = b, c, a + b + c
    return c

print(climb_three(5))       # 13
climb_opt.cpp
// 空间优化版：只需要两个变量
#include <iostream>
using namespace std;

int climb_stairs_o1(int n) {
    if (n <= 2) return n;
    int prev2 = 1, prev1 = 2;   // 分别代表 dp[i-2] 和 dp[i-1]
    for (int i = 3; i <= n; ++i) {
        int curr = prev1 + prev2;
        prev2 = prev1;
        prev1 = curr;
    }
    return prev1;
}

// 变形：每次可走 1、2、3 级
int climb_three(int n) {
    if (n == 1) return 1;
    if (n == 2) return 2;
    if (n == 3) return 4;   // 1+1+1, 1+2, 2+1, 3
    int a = 1, b = 2, c = 4;
    for (int i = 4; i <= n; ++i) {
        int nxt = a + b + c;
        a = b;
        b = c;
        c = nxt;
    }
    return c;
}

int main() {
    cout << climb_stairs_o1(10) << endl;  // 89
    cout << climb_three(5) << endl;       // 13
    return 0;
}
第 4 小节

综合实战：最小花费爬楼梯

带权重的状态转移

现在升级一下：每级台阶有一个"体力花费"，你可以从第 i 级花 cost[i] 的体力向上迈一步或两步。目标是爬到顶部（第 n 级之外），求最小花费。

定义 dp[i] 为到达第 i 级台阶所需的最小花费。状态转移方程变成：dp[i] = min(dp[i-1] + cost[i-1], dp[i-2] + cost[i-2])。含义是：最后一步要么从 i-1 迈上来，要么从 i-2 迈上来，取花费更小的那个。

min_cost_climb.py
# 最小花费爬楼梯
def min_cost_climbing_stairs(cost):
    n = len(cost)
    # dp[i] 表示到达第 i 级台阶的最小花费
    dp = [0] * (n + 1)
    dp[0] = 0
    dp[1] = 0
    for i in range(2, n + 1):
        dp[i] = min(dp[i - 1] + cost[i - 1],
                    dp[i - 2] + cost[i - 2])
    return dp[n]

cost = [10, 15, 20]
print(min_cost_climbing_stairs(cost))  # 15

cost2 = [1, 100, 1, 1, 1, 100, 1, 1, 100, 1]
print(min_cost_climbing_stairs(cost2)) # 6
min_cost_climb.cpp
// 最小花费爬楼梯
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int min_cost_climbing_stairs(vector<int>& cost) {
    int n = cost.size();
    // dp[i] 表示到达第 i 级台阶的最小花费
    vector<int> dp(n + 1, 0);
    dp[0] = 0;
    dp[1] = 0;
    for (int i = 2; i <= n; ++i) {
        dp[i] = min(dp[i - 1] + cost[i - 1],
                    dp[i - 2] + cost[i - 2]);
    }
    return dp[n];
}

int main() {
    vector<int> cost = {10, 15, 20};
    cout << min_cost_climbing_stairs(cost) << endl;  // 15

    vector<int> cost2 = {1, 100, 1, 1, 1, 100, 1, 1, 100, 1};
    cout << min_cost_climbing_stairs(cost2) << endl; // 6
    return 0;
}

这个实战题是 LeetCode 746 原题，也是很多竞赛的常客。它把"计数"变成了"求最值"，但核心仍然是状态转移。你可以试着自己改写成 O(1) 空间版本。
