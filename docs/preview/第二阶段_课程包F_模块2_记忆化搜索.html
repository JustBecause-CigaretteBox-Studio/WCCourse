第 1 小节

什么是记忆化搜索

记忆化搜索（Memoization Search）是动态规划最常见的实现方式之一。它先用递归把问题拆成小问题，再用一个"备忘录"把已经算过的结果存起来，避免重复计算。

你可以把它理解成：老师布置了一道很难的数学题，你请同学帮忙。第一个同学算了 10 分钟得出答案；第二个同学又问你同一道题，你直接把第一个同学的答案抄给他，不用再算 10 分钟。这个"抄答案"的过程，就是记忆化的精髓。

记忆化搜索特别适合思路自然、状态转移方程容易用递归表达的问题。因为它保留了递归的直观，同时获得了接近循环填表的效率。

memo_template.py
# 记忆化搜索的通用模板
memo = {}

def solve(state):
    # 1. 查表：如果算过，直接返回
    if state in memo:
        return memo[state]
    # 2. 边界条件
    if 边界条件:
        return 边界值
    # 3. 递归计算，并把结果存入 memo
    memo[state] = 递归计算结果
    return memo[state]
memo_template.cpp
// 记忆化搜索的通用模板
#include <iostream>
#include <vector>
using namespace std;

// 根据状态类型选择合适的数据结构
vector<int> memo;

int solve(int state) {
    // 1. 查表：如果算过，直接返回
    if (memo[state] != -1) return memo[state];
    // 2. 边界条件
    // if (边界条件) return 边界值;
    // 3. 递归计算，并把结果存入 memo
    // memo[state] = 递归计算结果;
    return memo[state];
}
第 2 小节

用记忆化改造递归

我们还是用斐波那契数列做例子，但这一次重点看"改造过程"。上面是普通递归，下面是记忆化版本。对比两者，你会发现只是多了三行代码：查表、存表、用字典用数组 / unordered_map。

这三行代码的威力非常大。当 n=40 时，普通递归需要几十亿次调用，浏览器或 Python 会明显卡顿；而记忆化版本只需要计算 40 个状态，几乎瞬间完成。

fib_compare.py
# 普通递归
 def fib(n):
     if n <= 1:
         return n
     return fib(n - 1) + fib(n - 2)

 # 记忆化搜索
 from functools import lru_cache

 @lru_cache(maxsize=None)   # Python 自带的记忆化装饰器
 def fib_fast(n):
     if n <= 1:
         return n
     return fib_fast(n - 1) + fib_fast(n - 2)

 print(fib_fast(100))        # 354224848179261915075
fib_compare.cpp
// 普通递归
// 计算 fib(5) 时，fib(2) 会被计算 3 次
#include <iostream>
#include <vector>
using namespace std;

int fib(int n) {
    if (n <= 1) return n;
    return fib(n - 1) + fib(n - 2);
}

// 记忆化搜索
vector<long long> memo;

long long fib_fast(int n) {
    if (memo[n] != -1) return memo[n];
    if (n <= 1) return n;
    memo[n] = fib_fast(n - 1) + fib_fast(n - 2);
    return memo[n];
}

int main() {
    memo.assign(101, -1);
    cout << fib_fast(100) << endl;  // 354224848179261915075
    return 0;
}

Python 标准库提供了 functools.lru_cache，它自动帮我们做记忆化。如果你不想自己维护字典，可以直接用它。但在学习阶段，建议先手写一遍字典版本，理解原理后再用装饰器偷懒。C++ 里可以自己维护一个 memo 数组（或 unordered_map）当备忘录。但在学习阶段，建议先手写一遍数组版本，理解原理后再用现成的数据结构偷懒。

第 3 小节

自顶向下 vs 自底向上

记忆化搜索是"自顶向下"：从大问题出发，需要什么子问题就递归去算。另一种思路是"自底向上"：先算最小的子问题，再一层层推到大问题。后者通常用循环和数组实现，也称为递推或填表。

两种写法在数学上是等价的，只是思考方向不同。记忆化搜索写起来更像递归，容易理解；自底向上写起来更像循环，常数更小，也更容易控制空间复杂度。

bottom_up.py
# 自底向上：循环填表
n = 10
dp = [0] * (n + 1)
dp[1] = 1                    # 边界
for i in range(2, n + 1):
    dp[i] = dp[i - 1] + dp[i - 2]
print(dp[n])                 # 55

# 空间优化：其实只需要前两个值
a, b = 0, 1
for _ in range(2, n + 1):
    a, b = b, a + b
print(b)                     # 55
bottom_up.cpp
// 自底向上：循环填表
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n = 10;
    vector<int> dp(n + 1, 0);
    dp[1] = 1;                    // 边界
    for (int i = 2; i <= n; ++i) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }
    cout << dp[n] << endl;        // 55

    // 空间优化：其实只需要前两个值
    int a = 0, b = 1;
    for (int i = 2; i <= n; ++i) {
        int nxt = a + b;
        a = b;
        b = nxt;
    }
    cout << b << endl;            // 55
    return 0;
}

初学时建议先掌握自顶向下的记忆化搜索，因为它保留了问题的原始结构；遇到性能瓶颈时，再考虑改成自底向上或空间优化版本。

第 4 小节

综合实战：数字三角形

从递归到记忆化

有一个数字三角形，从顶部出发，每次只能向左下或右下走，求到达底部时的最大路径和。这个问题非常适合用记忆化搜索：到达某个位置的最大和，等于它上方两个位置最大和中的较大者，加上当前数字。

定义 dfs(r, c) 为从第 r 行第 c 列出发到底部的最大和。转移方程是：dfs(r, c) = triangle[r][c] + max(dfs(r+1, c), dfs(r+1, c+1))。用字典或二维列表做备忘录用二维数组或 unordered_map 做备忘录即可。

triangle.py
# 数字三角形：记忆化搜索版
triangle = [
    [7],
    [3, 8],
    [8, 1, 0],
    [2, 7, 4, 4],
    [4, 5, 2, 6, 5]
]

memo = {}

def dfs(r, c):
    if r == len(triangle) - 1:
        return triangle[r][c]
    if (r, c) in memo:
        return memo[(r, c)]
    memo[(r, c)] = triangle[r][c] + max(dfs(r + 1, c), dfs(r + 1, c + 1))
    return memo[(r, c)]

print(dfs(0, 0))   # 30（路径 7->3->8->7->5）
triangle.cpp
// 数字三角形：记忆化搜索版
#include <iostream>
#include <vector>
using namespace std;

vector<vector<int>> triangle = {
    {7},
    {3, 8},
    {8, 1, 0},
    {2, 7, 4, 4},
    {4, 5, 2, 6, 5}
};

vector<vector<int>> memo;

int dfs(int r, int c) {
    if (r == (int)triangle.size() - 1) return triangle[r][c];
    if (memo[r][c] != -1) return memo[r][c];
    memo[r][c] = triangle[r][c] + max(dfs(r + 1, c), dfs(r + 1, c + 1));
    return memo[r][c];
}

int main() {
    memo.assign(triangle.size(), vector<int>(triangle.back().size(), -1));
    cout << dfs(0, 0) << endl;   // 30（路径 7->3->8->7->5）
    return 0;
}

这个例子展示了记忆化搜索处理二维状态的典型写法：dfs(r, c) 有两个参数，备忘录用 (r, c) 元组当键memo[r][c] 二维数组下标当键。试着把 triangle 换成你自己的数据，看看最大和会不会变化。
