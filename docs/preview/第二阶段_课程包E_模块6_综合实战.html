第 1 小节

最短路径问题

从地图导航到网络路由

生活中到处都有最短路径问题：导航软件帮你找最快的路线、快递规划最省油的配送路径、网络数据包选择延迟最小的链路。这些问题都可以用图来建模。

在带权图中，每条边有一个权重（例如距离、时间、费用），我们要找的是从一个起点到终点的总权重最小的路径。

图：一张带权图。边上的数字是权重（可以想象成距离、时间或费用），箭头表示只能顺着箭头方向走。蓝色加粗的边是从 0 到 3 的最短路径：0 → 2 → 1 → 3，总权重 1 + 2 + 1 = 4；看起来更直接的 0 → 1 → 3 要花 4 + 1 = 5，反而更远。第 3 小节的代码用的就是这张图。

解决单源最短路径问题最著名的算法之一就是 Dijkstra 算法，它的前提是所有边的权重都不能为负数。

第 2 小节

Dijkstra 的核心思想

每次选当前最近的未确定点

Dijkstra 算法的思路很像「贪心（每次只找最多或最少的）」：从起点开始，维护每个节点到起点的当前最短距离。每次从所有还没确定最短距离的节点中，选一个距离最小的节点，把它标记为已确定；然后用它去更新邻居的距离。

可以把它想象成往水里扔石头：水波一圈一圈向外扩散，水波最先到达某个地方的路径是最短的。

算法需要的数据结构包括：一个记录最短距离的数组 dist，一个记录节点是否已经确定最短距离的 visited 集合。

第 3 小节

实现 Dijkstra

用邻接表和优先队列优化

下面是一个用 Python 实现的简化版 Dijkstra，图用邻接表表示，节点用数字编号。每次选择当前距离最小的未访问节点，并更新其邻居。

import heapq

def dijkstra(graph, start):
    # graph[u] = [(v, weight), ...]
    dist = {node: float('inf') for node in graph}
    dist[start] = 0
    heap = [(0, start)]

    while heap:
        d, u = heapq.heappop(heap)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(heap, (dist[v], v))
    return dist

graph = {
    0: [(1, 4), (2, 1)],
    1: [(3, 1)],
    2: [(1, 2), (3, 5)],
    3: []
}
print(dijkstra(graph, 0))   # {0: 0, 1: 3, 2: 1, 3: 4}

下面是一个用 C++ 实现的简化版 Dijkstra，图用邻接表表示，节点用数字编号。每次选择当前距离最小的未访问节点，并更新其邻居。

dijkstra.cpp
#include <vector>
#include <queue>
#include <limits>
#include <iostream>
using namespace std;

using P = pair<int, int>;  // (distance, node)

vector<int> dijkstra(const vector<vector<pair<int,int>>>& graph, int start) {
    int n = graph.size();
    vector<int> dist(n, numeric_limits<int>::max());
    dist[start] = 0;
    priority_queue<P, vector<P>, greater<P>> heap;
    heap.push({0, start});

    while (!heap.empty()) {
        auto [d, u] = heap.top();
        heap.pop();
        if (d > dist[u]) continue;
        for (auto [v, w] : graph[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                heap.push({dist[v], v});
            }
        }
    }
    return dist;
}

int main() {
    vector<vector<pair<int,int>>> graph = {
        {{1, 4}, {2, 1}},
        {{3, 1}},
        {{1, 2}, {3, 5}},
        {}
    };
    auto dist = dijkstra(graph, 0);
    for (int d : dist) cout << d << " ";   // 0 3 1 4
    return 0;
}

这里用到了优先队列（堆），它能快速取出当前距离最小的节点，让算法更高效。

第 4 小节

综合实战：城市最短路线

规划从家到学校的最短路径

假设有几座城市和它们之间的距离，用 Dijkstra 算法找出从「家」到「学校」的最短路线。我们不仅让程序算出最短距离，还可以额外用一个 prev 数组记录每个节点的前驱，最后倒推得到完整路径。

import heapq

def shortest_path(graph, start, end):
    dist = {node: float('inf') for node in graph}
    prev = {node: None for node in graph}
    dist[start] = 0
    heap = [(0, start)]

    while heap:
        d, u = heapq.heappop(heap)
        if u == end:
            break
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                prev[v] = u
                heapq.heappush(heap, (dist[v], v))

    path = []
    cur = end
    while cur is not None:
        path.append(cur)
        cur = prev[cur]
    path.reverse()
    return dist[end], path

graph = {
    "家": [("路口A", 2), ("路口B", 4)],
    "路口A": [("学校", 7)],
    "路口B": [("路口C", 1)],
    "路口C": [("学校", 2)],
    "学校": []
}
print(shortest_path(graph, "家", "学校"))   # (7, ['家', '路口B', '路口C', '学校'])
city_route.cpp
#include <queue>
#include <limits>
#include <unordered_map>
#include <vector>
#include <string>
#include <iostream>
#include <algorithm>
using namespace std;

using P = pair<int, string>;  // (distance, node)

pair<int, vector<string>> shortest_path(
    const unordered_map<string, vector<pair<string,int>>>& graph,
    const string& start,
    const string& end) {

    unordered_map<string, int> dist;
    unordered_map<string, string> prev;
    for (auto& [node, _] : graph) dist[node] = numeric_limits<int>::max();
    dist[start] = 0;

    priority_queue<P, vector<P>, greater<P>> heap;
    heap.push({0, start});

    while (!heap.empty()) {
        auto [d, u] = heap.top();
        heap.pop();
        if (u == end) break;
        for (auto [v, w] : graph.at(u)) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                prev[v] = u;
                heap.push({dist[v], v});
            }
        }
    }

    vector<string> path;
    string cur = end;
    while (!cur.empty()) {
        path.push_back(cur);
        auto it = prev.find(cur);
        if (it == prev.end()) break;
        cur = it->second;
    }
    reverse(path.begin(), path.end());
    return {dist[end], path};
}

int main() {
    unordered_map<string, vector<pair<string,int>>> graph = {
        {"家", {{"路口A", 2}, {"路口B", 4}}},
        {"路口A", {{"学校", 7}}},
        {"路口B", {{"路口C", 1}}},
        {"路口C", {{"学校", 2}}},
        {"学校", {}}
    };
    auto [d, path] = shortest_path(graph, "家", "学校");
    cout << d << "\n";
    for (auto& s : path) cout << s << " ";
    return 0;
}
