第 1 小节

认识 0-1 背包

背包问题是动态规划中最经典的模型之一。问题场景很好理解：你有一个容量为 W 的背包，还有 n 件物品，每件物品有重量 wi 和价值 vi。每件物品要么完整放进背包（1 件），要么不放（0 件），不能拆分。问在不超过背包容量的前提下，能装下的最大价值是多少？

这就是"0-1 背包"名字的由来——每件物品只有"拿"或"不拿"两种选择。它和现实生活很像：旅行箱空间有限，你要决定带哪些东西才能让总价值最高。

暴力解法是枚举所有物品的选法，共有 2^n 种，显然太慢。动态规划可以把时间复杂度降到 O(n×W)，关键就是定义好状态和转移方程。

# 0-1 背包问题描述

# weight[i] 表示第 i 件物品的重量
# value[i] 表示第 i 件物品的价值

第 2 小节

二维 DP 与状态转移

定义 dp[i][j] 表示：只考虑前 i 件物品，背包容量为 j 时能获得的最大价值。这里 i 通常从 0 到 n，j 从 0 到 W。

对于第 i 件物品，只有两种选择：
① 不拿：最大价值等于 dp[i-1][j]（前 i-1 件物品、容量 j 的最优值）。
② 拿：前提是容量足够（j >= weight[i]），价值等于 dp[i-1][j-weight[i]] + value[i]。

状态转移方程就是：dp[i][j] = max(dp[i-1][j], dp[i-1][j-weight[i]] + value[i])，前提是 j 够大；否则只能不拿。

knapsack_2d.py
# 0-1 背包：二维 DP 版本
def knapsack_2d(weights, values, W):
    n = len(weights)
    dp = [[0] * (W + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(W + 1):
            # 不拿第 i-1 件物品
            dp[i][j] = dp[i - 1][j]
            # 拿第 i-1 件物品
            if j >= weights[i - 1]:
                dp[i][j] = max(dp[i][j],
                               dp[i - 1][j - weights[i - 1]] + values[i - 1])
    return dp[n][W]

weights = [2, 3, 4]
values  = [3, 4, 5]
W = 5
print(knapsack_2d(weights, values, W))  # 7
knapsack_2d.cpp
// 0-1 背包：二维 DP 版本
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int knapsack_2d(vector<int>& weights, vector<int>& values, int W) {
    int n = weights.size();
    vector<vector<int>> dp(n + 1, vector<int>(W + 1, 0));
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j <= W; ++j) {
            // 不拿第 i-1 件物品
            dp[i][j] = dp[i - 1][j];
            // 拿第 i-1 件物品
            if (j >= weights[i - 1]) {
                dp[i][j] = max(dp[i][j],
                               dp[i - 1][j - weights[i - 1]] + values[i - 1]);
            }
        }
    }
    return dp[n][W];
}

int main() {
    vector<int> weights = {2, 3, 4};
    vector<int> values  = {3, 4, 5};
    int W = 5;
    cout << knapsack_2d(weights, values, W) << endl;  // 7
    return 0;
}
第 3 小节

滚动数组优化

观察二维 DP 的状态转移，计算第 i 行时只依赖第 i-1 行。因此我们可以把二维数组压缩成一维数组，用 dp[j] 表示容量为 j 时的最大价值。

关键细节：更新一维数组时，容量 j 必须从大到小遍历。如果从小到大，同一轮中先更新的 dp[j-weight] 会被当成"本轮已经拿过"的状态，导致某件物品被重复选取，那就变成"完全背包"了。

knapsack_1d.py
# 0-1 背包：一维滚动数组优化
def knapsack_1d(weights, values, W):
    n = len(weights)
    dp = [0] * (W + 1)
    for i in range(n):
        # 必须倒序遍历容量
        for j in range(W, weights[i] - 1, -1):
            dp[j] = max(dp[j],
                        dp[j - weights[i]] + values[i])
    return dp[W]

weights = [2, 3, 4]
values  = [3, 4, 5]
W = 5
print(knapsack_1d(weights, values, W))  # 7

# 错误示范：正序遍历会导致重复选取
# for j in range(weights[i], W + 1):
#     dp[j] = max(dp[j], dp[j - weights[i]] + values[i])  # 这是完全背包！
knapsack_1d.cpp
// 0-1 背包：一维滚动数组优化
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int knapsack_1d(vector<int>& weights, vector<int>& values, int W) {
    int n = weights.size();
    vector<int> dp(W + 1, 0);
    for (int i = 0; i < n; ++i) {
        // 必须倒序遍历容量
        for (int j = W; j >= weights[i]; --j) {
            dp[j] = max(dp[j],
                        dp[j - weights[i]] + values[i]);
        }
    }
    return dp[W];
}

int main() {
    vector<int> weights = {2, 3, 4};
    vector<int> values  = {3, 4, 5};
    int W = 5;
    cout << knapsack_1d(weights, values, W) << endl;  // 7

    // 错误示范：正序遍历会导致重复选取
    // for (int j = weights[i]; j <= W; ++j)
    //     dp[j] = max(dp[j], dp[j - weights[i]] + values[i]);  // 这是完全背包！
    return 0;
}
第 4 小节

综合实战：恰好装满的背包

0-1 背包变形

在很多题目中，不是"不超过容量"，而是"恰好装满容量"。比如问：能否选出若干件物品，使总重量恰好等于 W？这时 DP 数组的初始化要做调整。

把 dp[0] 初始化为 0（容量为 0 时价值为 0），其他 dp[j] 初始化为负无穷或一个特殊值，表示"还没法恰好凑成"。转移方程和 0-1 背包一样，只是最后看 dp[W] 是否被更新过。

exact_knapsack.py
# 恰好装满：判断能否凑出重量 W
INF = float('-inf')

def can_fill_exactly(weights, W):
    dp = [INF] * (W + 1)
    dp[0] = 0
    for w in weights:
        for j in range(W, w - 1, -1):
            if dp[j - w] != INF:
                dp[j] = max(dp[j], dp[j - w] + w)  # 这里价值等于重量本身
    return dp[W] != INF

weights = [2, 3, 5, 7]
print(can_fill_exactly(weights, 10))  # True  (2+3+5 或 3+7)
print(can_fill_exactly(weights, 11))  # False (没有组合能恰好凑出 11)
exact_knapsack.cpp
// 恰好装满：判断能否凑出重量 W
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

bool can_fill_exactly(vector<int>& weights, int W) {
    const int INF = -1e9;
    vector<int> dp(W + 1, INF);
    dp[0] = 0;
    for (int w : weights) {
        for (int j = W; j >= w; --j) {
            if (dp[j - w] != INF) {
                dp[j] = max(dp[j], dp[j - w] + w);  // 这里价值等于重量本身
            }
        }
    }
    return dp[W] != INF;
}

int main() {
    vector<int> weights = {2, 3, 5, 7};
    cout << (can_fill_exactly(weights, 10) ? "True" : "False") << endl;  // True  (2+3+5 或 3+7)
    cout << (can_fill_exactly(weights, 11) ? "True" : "False") << endl;  // False (没有组合能恰好凑出 11)
    return 0;
}

这个实战帮你体会 0-1 背包的"选择的艺术"：每件物品拿与不拿，组合起来就能回答很多有趣的问题。你可以把上面的代码改成输出具体方案，或者改成求"恰好装满时的最小物品数"。
