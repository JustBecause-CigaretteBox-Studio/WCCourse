第 1 小节

最长上升子序列（LIS）

最长上升子序列（Longest Increasing Subsequence，LIS）问题是：给定一个序列，找出其中最长的严格递增子序列的长度。子序列不要求连续，但要求相对顺序不变。

例如序列 [10, 9, 2, 5, 3, 7, 101, 18] 的最长上升子序列是 [2, 3, 7, 101] 或 [2, 5, 7, 101]，长度为 4。

定义 dp[i] 为：以第 i 个数结尾的最长上升子序列长度。那么对于每个 i，去看前面所有比它小的数 j，有 dp[i] = max(dp[i], dp[j] + 1)。边界是 dp[i] = 1（每个数自己就是一个长度为 1 的上升子序列）。

lis.py
# 最长上升子序列：O(n^2) DP
def length_of_lis(nums):
    if not nums:
        return 0
    n = len(nums)
    dp = [1] * n
    for i in range(1, n):
        for j in range(i):
            if nums[j] < nums[i]:
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp)

nums = [10, 9, 2, 5, 3, 7, 101, 18]
print(length_of_lis(nums))  # 4
lis.cpp
// 最长上升子序列：O(n^2) DP
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int length_of_lis(vector<int>& nums) {
    if (nums.empty()) return 0;
    int n = nums.size();
    vector<int> dp(n, 1);
    for (int i = 1; i < n; ++i) {
        for (int j = 0; j < i; ++j) {
            if (nums[j] < nums[i]) {
                dp[i] = max(dp[i], dp[j] + 1);
            }
        }
    }
    return *max_element(dp.begin(), dp.end());
}

int main() {
    vector<int> nums = {10, 9, 2, 5, 3, 7, 101, 18};
    cout << length_of_lis(nums) << endl;  // 4
    return 0;
}
第 2 小节

最长公共子序列（LCS）

最长公共子序列（Longest Common Subsequence，LCS）问题是：给定两个序列，求它们最长的公共子序列长度。子序列同样不要求连续。

例如 "abcde" 和 "ace" 的 LCS 是 "ace"，长度为 3。"abc" 和 "def" 的 LCS 长度为 0。

定义 dp[i][j] 为：text1 的前 i 个字符和 text2 的前 j 个字符的 LCS 长度。转移时比较两个串的最后一个字符：如果相同，则 dp[i][j] = dp[i-1][j-1] + 1；如果不同，则 dp[i][j] = max(dp[i-1][j], dp[i][j-1])。

lcs.py
# 最长公共子序列：二维 DP
def longest_common_subsequence(text1, text2):
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i - 1] == text2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1] + 1
            else:
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1])
    return dp[m][n]

print(longest_common_subsequence("abcde", "ace"))  # 3
print(longest_common_subsequence("abc", "def"))    # 0
lcs.cpp
// 最长公共子序列：二维 DP
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

int longest_common_subsequence(string text1, string text2) {
    int m = text1.size(), n = text2.size();
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    for (int i = 1; i <= m; ++i) {
        for (int j = 1; j <= n; ++j) {
            if (text1[i - 1] == text2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1;
            } else {
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
            }
        }
    }
    return dp[m][n];
}

int main() {
    cout << longest_common_subsequence("abcde", "ace") << endl;  // 3
    cout << longest_common_subsequence("abc", "def") << endl;    // 0
    return 0;
}
第 3 小节

回顾与练习

LIS 和 LCS 是最常见的线性 DP 模型。它们的共同点是：把"以某个位置结尾"或"前若干个字符"作为状态，通过比较前面已经算出的状态来推导当前状态。

练习：把 LIS 的代码改成输出一条具体的最长上升子序列（而不仅是长度）。提示：在更新 dp[i] 时，同时记录 prev[i] 表示 i 是从哪个 j 转移过来的，最后从最大值位置反向追踪。

lis_path.py
# 练习：输出 LIS 的一条路径
def lis_with_path(nums):
    n = len(nums)
    dp = [1] * n
    prev = [-1] * n
    for i in range(n):
        for j in range(i):
            if nums[j] < nums[i] and dp[j] + 1 > dp[i]:
                dp[i] = dp[j] + 1
                prev[i] = j
    # 找终点
    best = max(range(n), key=lambda i: dp[i])
    # 反向追踪
    path = []
    cur = best
    while cur != -1:
        path.append(nums[cur])
        cur = prev[cur]
    path.reverse()
    return dp[best], path

print(lis_with_path([10, 9, 2, 5, 3, 7, 101, 18]))
# 输出：(4, [2, 3, 7, 101])
lis_path.cpp
// 练习：输出 LIS 的一条路径
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

pair<int, vector<int>> lis_with_path(vector<int>& nums) {
    int n = nums.size();
    vector<int> dp(n, 1);
    vector<int> prev(n, -1);
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < i; ++j) {
            if (nums[j] < nums[i] && dp[j] + 1 > dp[i]) {
                dp[i] = dp[j] + 1;
                prev[i] = j;
            }
        }
    }
    // 找终点
    int best = 0;
    for (int i = 1; i < n; ++i) {
        if (dp[i] > dp[best]) best = i;
    }
    // 反向追踪
    vector<int> path;
    int cur = best;
    while (cur != -1) {
        path.push_back(nums[cur]);
        cur = prev[cur];
    }
    reverse(path.begin(), path.end());
    return {dp[best], path};
}

int main() {
    vector<int> nums = {10, 9, 2, 5, 3, 7, 101, 18};
    auto res = lis_with_path(nums);
    cout << res.first << endl;
    for (int x : res.second) cout << x << " ";
    cout << endl;
    // 输出：4
    //       2 3 7 101
    return 0;
}
第 4 小节

综合实战：编辑距离

线性 DP 的拓展

编辑距离（Edit Distance）是 LCS 的近亲，也是面试和竞赛高频题。题目：把字符串 word1 转换成 word2，最少需要多少次操作？每次可以插入、删除或替换一个字符。

定义 dp[i][j] 为 word1 前 i 个字符变成 word2 前 j 个字符的最少操作数。如果 word1[i-1] == word2[j-1]，则不需要额外操作：dp[i][j] = dp[i-1][j-1]；否则取插入、删除、替换三种操作的最小值加 1。

edit_distance.py
# 编辑距离
def min_distance(word1, word2):
    m, n = len(word1), len(word2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    # 边界：一个空串变成另一个串
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if word1[i - 1] == word2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]
            else:
                dp[i][j] = min(dp[i - 1][j],      # 删除
                               dp[i][j - 1],      # 插入
                               dp[i - 1][j - 1]) + 1  # 替换
    return dp[m][n]

print(min_distance("horse", "ros"))    # 3
print(min_distance("intention", "execution"))  # 5
edit_distance.cpp
// 编辑距离
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
using namespace std;

int min_distance(string word1, string word2) {
    int m = word1.size(), n = word2.size();
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    // 边界：一个空串变成另一个串
    for (int i = 0; i <= m; ++i) dp[i][0] = i;
    for (int j = 0; j <= n; ++j) dp[0][j] = j;
    for (int i = 1; i <= m; ++i) {
        for (int j = 1; j <= n; ++j) {
            if (word1[i - 1] == word2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1];
            } else {
                dp[i][j] = min({dp[i - 1][j],      // 删除
                                dp[i][j - 1],      // 插入
                                dp[i - 1][j - 1]}) + 1;  // 替换
            }
        }
    }
    return dp[m][n];
}

int main() {
    cout << min_distance("horse", "ros") << endl;           // 3
    cout << min_distance("intention", "execution") << endl; // 5
    return 0;
}

编辑距离把线性 DP 从"最长"问题拓展到"最省代价"问题。它依然遵循"先定义状态、再写转移、最后处理边界"的三步套路。熟练掌握这个套路，大多数线性 DP 都能迎刃而解。
